JavaScript Case Transformation Regex Helpers

const toCamelCase = s => s.toLowerCase().replace(/[^a-zA-Z0-9]+(.)/g, (m, chr) => chr.toUpperCase());
const toSnakeCase = s => s.match(/[A-Z]{2,}(?=[A-Z][a-z]+[0-9]*|\b)|[A-Z]?[a-z]+[0-9]*|[A-Z]|[0-9]+/g)
                          .map(x => x.toLowerCase()).join('_');
const toKebabCase = s => toSnakeCase(s).replace(/_/g, '-');
